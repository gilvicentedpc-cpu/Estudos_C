#include <unistd.h>

unsigned int ft_strlcpy(char *dest, char *src, unsigned int size)
{
    unsigned int i;
    i = 0;

    while(src[i] < size - 1)
    {
        dest = src;
        i++;    
    }
    return (*dest);
}

{
    char dest[50] = "42 ";
