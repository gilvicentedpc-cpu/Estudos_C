#include <stdio.h>

int ft_is_prime(int nb)
{
   int i;

   if (nb <= 0)
            return(0);

   i = 2;
   while (i <= nb / 2)
   {
        if (nb % i == 0)
        {
        return (0);
        }
   i++;
   }
   return (1);
}

int ft_find_next_prime(int nb)
{
    if (ft_is_prime(nb) == 0)
    {
        if (nb % 2 == 0)
        {
          nb++;  
        }
    }
    if (ft_is_prime(nb) == 1)
    {
        nb += 2;
    }
    return (nb);
}

int main(void)
{
    int i;
    int result;

    result = ft_find_next_prime(i);
    printf ("%d\n", result);
